cmake_minimum_required(VERSION 3.16)
project(harfang_web_asset_bridge LANGUAGES CXX)
option(HG_WEB_BUILD_BRIDGE "Build the historical HARFANG reader bridge" ON)
set(HARFANG_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../harfang3d" CACHE PATH "HARFANG source checkout (build dependency only)")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/adapters/default.vert" DEFAULT_VERTEX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/adapters/default.frag" DEFAULT_FRAGMENT)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/adapters/depth.vert" DEPTH_VERTEX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/adapters/depth.frag" DEPTH_FRAGMENT)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/adapters/source-hashes.json" APPROVED_SOURCES)
configure_file(adapters/embedded.h.in generated/embedded.h @ONLY)
add_executable(assetc-web assetc_web.cpp)
target_compile_features(assetc-web PRIVATE cxx_std_17)
target_include_directories(assetc-web PRIVATE "${HARFANG_SOURCE}/extern" "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_property(TARGET assetc-web PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(assetc-web PRIVATE /utf-8 /W4)
endif()
install(TARGETS assetc-web RUNTIME DESTINATION .)
if(HG_WEB_BUILD_BRIDGE)
add_executable(harfang_web_asset_bridge asset_bridge.cpp)
target_compile_features(harfang_web_asset_bridge PRIVATE cxx_std_17)
if(TARGET engine)
  target_link_libraries(harfang_web_asset_bridge PRIVATE engine)
elseif(HARFANG_LINK_CONFIG)
  include("${HARFANG_LINK_CONFIG}")
  target_include_directories(harfang_web_asset_bridge PRIVATE ${HG_BRIDGE_INCLUDES})
  target_compile_definitions(harfang_web_asset_bridge PRIVATE ${HG_BRIDGE_DEFINITIONS})
  target_link_libraries(harfang_web_asset_bridge PRIVATE ${HG_BRIDGE_LIBRARIES})
  set_property(TARGET harfang_web_asset_bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
else()
  message(FATAL_ERROR "Add this directory to a HARFANG build with target engine, or use tools/build_native.py with an existing MSVC assetc build.")
endif()
endif()
